};

\node[dag_result, text width=5.4cm] (Lam) at (-5.5,-7.4) {

\node[dag_result, text width=5.6cm] (P2) at (2.5,-7.4) {

};

\node[dag_result, text width=5.4cm] (Prod) at (10.5,-7.4) {
  \textbf{Proposition 3: productivity effect $>0$} \\
  $((W/\gamma)^{1-\hat\sigma}-R^{1-\hat\sigma})/(1-\hat\sigma)>0$\\
  when $W/\gamma(I^*)>R$, $\hat\sigma\ne1$
};

\node[dag_template_note, text width=5.6cm] (Ext) at (18.5,-7.4) {

  wage rises iff $K/L>e^{1/(N-I^*)}$ ---\\
  the printed direction of $\bar K$ is reversed
};
